/* FILE 2: js/process_manager.js */
class PCB {
    constructor(pid, name, priority, burstTime, memoryRequired = 64, resourceClaim = 'none') {
        this.pid = pid;
        this.name = name;
        this.priority = parseInt(priority);
        this.burstTime = parseInt(burstTime);
        this.remainingTime = parseInt(burstTime);
        this.memoryRequired = parseInt(memoryRequired);
        this.resourceClaim = resourceClaim;
        this.state = 'NEW'; 
        this.arrivalTime = 0;
        this.startTime = -1;
        this.completionTime = 0;
        this.waitTime = 0;
        this.turnaroundTime = 0;
        this.age = 0;
        this.color = this.assignColor(this.priority);
    }

    assignColor(prio) {
        switch(prio) {
            case 1: return '#ff4757';
            case 2: return '#ffa502';
            case 3: return '#00d2d3';
            case 4: return '#2ed573';
            default: return '#00f2fe';
        }
    }

    getCategoryName() {
        switch(this.priority) {
            case 1: return 'Emergency Trauma';
            case 2: return 'Urgent Surgery';
            case 3: return 'General Ward';
            case 4: return 'Routine Checkup';
            default: return 'Standard Case';
        }
    }
}

class ProcessManager {
    constructor() {
        this.processes = [];
        this.nextPidCounter = 101;
    }

    createProcess(name, priority, burstTime, memoryRequired, resourceClaim) {
        const pid = `P${this.nextPidCounter++}`;
        const proc = new PCB(pid, name, priority, burstTime, memoryRequired, resourceClaim);
        this.processes.push(proc);
        return proc;
    }

    getProcess(pid) {
        return this.processes.find(p => p.pid === pid);
    }

    getAllProcesses() {
        return this.processes;
    }

    getProcessesByState(state) {
        return this.processes.filter(p => p.state === state);
    }

    resetAll() {
        this.processes = [];
        this.nextPidCounter = 101;
    }
}

window.processManager = new ProcessManager();

class CPUScheduler {
    constructor() {
        this.algorithm = 'priority_p';
        this.timeQuantum = 2;
        this.rrTickCounter = 0;
        this.runningProcess = null;
        this.ganttHistory = [];
        this.contextSwitches = 0;
        this.totalBusyTicks = 0;
    }

    setAlgorithm(algo) {
        this.algorithm = algo;
        this.rrTickCounter = 0;
    }

    scheduleNext(currentTick, readyQueue, processManager) {
        readyQueue.forEach(proc => {
            proc.age++;
            if (proc.age > 10 && proc.priority > 1) {
                proc.effectivePriority = Math.max(1, proc.priority - 1);
            } else {
                proc.effectivePriority = proc.priority;
            }
        });

        let selectedProc = null;

        if (readyQueue.length > 0) {
            if (this.algorithm === 'priority_p') {
                readyQueue.sort((a, b) => {
                    const prioA = a.effectivePriority !== undefined ? a.effectivePriority : a.priority;
                    const priob = b.effectivePriority !== undefined ? b.effectivePriority : b.priority;
                    if (prioA !== priob) return prioA - priob;
                    return a.pid.localeCompare(b.pid);
                });
                selectedProc = readyQueue[0];

            } else if (this.algorithm === 'sjf') {
                readyQueue.sort((a, b) => a.remainingTime - b.remainingTime);
                selectedProc = readyQueue[0];

            } else if (this.algorithm === 'rr') {
                if (this.runningProcess && readyQueue.includes(this.runningProcess) && this.rrTickCounter < this.timeQuantum) {
                    selectedProc = this.runningProcess;
                } else {
                    this.rrTickCounter = 0;
                    selectedProc = readyQueue[0];
                }

            } else if (this.algorithm === 'fcfs') {
                readyQueue.sort((a, b) => a.arrivalTime - b.arrivalTime);
                selectedProc = readyQueue[0];
            }
        }

        if (selectedProc !== this.runningProcess) {
            if (this.runningProcess && this.runningProcess.state === 'RUNNING') {
                this.runningProcess.state = 'READY';
                window.uiController.logSystem(`[PREEMPTION] Process ${this.runningProcess.pid} (${this.runningProcess.name}) preempted by ${selectedProc ? selectedProc.pid : 'IDLE'}`, 'preempt');
            }

            if (selectedProc) {
                this.contextSwitches++;
                this.rrTickCounter = 0;
                selectedProc.state = 'RUNNING';
                if (selectedProc.startTime === -1) {
                    selectedProc.startTime = currentTick;
                }
                window.uiController.logSystem(`[CPU ALLOCATION] ${selectedProc.pid} (${selectedProc.name}) scheduled on CPU Core under ${this.algorithm.toUpperCase()}`, 'schedule');
            }

            this.runningProcess = selectedProc;
        } else if (this.runningProcess) {
            this.rrTickCounter++;
        }

        if (this.runningProcess) {
            this.totalBusyTicks++;
            this.runningProcess.remainingTime--;
            this.runningProcess.age = 0;

            this.recordGanttStep(currentTick, this.runningProcess);

            if (this.runningProcess.remainingTime <= 0) {
                this.runningProcess.state = 'TERMINATED';
                this.runningProcess.completionTime = currentTick + 1;
                this.runningProcess.turnaroundTime = this.runningProcess.completionTime - this.runningProcess.arrivalTime;
                this.runningProcess.waitTime = this.runningProcess.turnaroundTime - this.runningProcess.burstTime;

                window.uiController.logSystem(`[PROCESS DISCHARGED] ${this.runningProcess.pid} (${this.runningProcess.name}) completed treatment! Wait Time: ${this.runningProcess.waitTime}s, Turnaround: ${this.runningProcess.turnaroundTime}s`, 'system');

                if (window.syncEngine) {
                    window.syncEngine.releaseResourceByPID(this.runningProcess.pid);
                }
                if (window.memoryManager) {
                    window.memoryManager.deallocateMemory(this.runningProcess.pid);
                }

                this.runningProcess = null;
                this.rrTickCounter = 0;
            }
        } else {
            this.recordGanttStep(currentTick, null);
        }
    }

    recordGanttStep(tick, proc) {
        const lastBlock = this.ganttHistory[this.ganttHistory.length - 1];
        const pid = proc ? proc.pid : 'IDLE';
        const color = proc ? proc.color : 'rgba(255,255,255,0.05)';
        const priority = proc ? proc.priority : 0;
        const name = proc ? proc.name : 'Idle';

        if (lastBlock && lastBlock.pid === pid) {
            lastBlock.endTick = tick + 1;
            lastBlock.duration = lastBlock.endTick - lastBlock.startTick;
        } else {
            this.ganttHistory.push({
                pid: pid,
                name: name,
                priority: priority,
                color: color,
                startTick: tick,
                endTick: tick + 1,
                duration: 1
            });
        }
    }

    reset() {
        this.runningProcess = null;
        this.ganttHistory = [];
        this.contextSwitches = 0;
        this.totalBusyTicks = 0;
        this.rrTickCounter = 0;
    }
}

class BankerAndRAG {
    constructor() {
        this.resources = {
            'ICU': { total: 3, available: 3 },
            'Ventilator': { total: 2, available: 2 },
            'MRI': { total: 2, available: 2 },
            'OR': { total: 2, available: 2 }
        };
        this.resourceKeys = Object.keys(this.resources);
        this.lastSafetyResult = null;
    }

    checkSafetyState(processes) {
        const activeProcs = processes.filter(p => p.state !== 'TERMINATED');
        const numProcs = activeProcs.length;
        if (numProcs === 0) {
            return { safe: true, sequence: [], message: 'System is SAFE (No active processes)' };
        }

        const work = {};
        this.resourceKeys.forEach(res => {
            work[res] = this.resources[res].available;
        });

        const finish = new Array(numProcs).fill(false);
        const safeSeq = [];

        let count = 0;
        let progress = true;

        while (count < numProcs && progress) {
            progress = false;
            for (let i = 0; i < numProcs; i++) {
                if (!finish[i]) {
                    const proc = activeProcs[i];
                    const need = this.getProcessNeed(proc);
                    
                    let canExecute = true;
                    for (const res of this.resourceKeys) {
                        if ((need[res] || 0) > work[res]) {
                            canExecute = false;
                            break;
                        }
                    }

                    if (canExecute) {
                        const alloc = this.getProcessAllocation(proc);
                        for (const res of this.resourceKeys) {
                            work[res] += (alloc[res] || 0);
                        }
                        finish[i] = true;
                        safeSeq.push(proc.pid);
                        progress = true;
                        count++;
                    }
                }
            }
        }

        if (count === numProcs) {
            this.lastSafetyResult = {
                safe: true,
                sequence: safeSeq,
                message: `System is in a SAFE State! Safe Execution Sequence: ${safeSeq.join(' -> ')}`
            };
        } else {
            this.lastSafetyResult = {
                safe: false,
                sequence: [],
                message: `UNSAFE STATE / DEADLOCK RISK! Resource contention preventing safe sequence.`
            };
        }

        return this.lastSafetyResult;
    }

    getProcessAllocation(proc) {
        const alloc = { ICU: 0, Ventilator: 0, MRI: 0, OR: 0 };
        if (proc.resourceClaim && proc.resourceClaim !== 'none') {
            if (proc.state === 'RUNNING' || proc.state === 'WAITING' || proc.state === 'READY') {
                if (window.syncEngine && window.syncEngine.isHolder(proc.resourceClaim, proc.pid)) {
                    alloc[proc.resourceClaim] = 1;
                }
            }
        }
        return alloc;
    }

    getProcessMax(proc) {
        const max = { ICU: 0, Ventilator: 0, MRI: 0, OR: 0 };
        if (proc.resourceClaim && proc.resourceClaim !== 'none') {
            max[proc.resourceClaim] = 1;
        }
        return max;
    }

    getProcessNeed(proc) {
        const max = this.getProcessMax(proc);
        const alloc = this.getProcessAllocation(proc);
        const need = {};
        this.resourceKeys.forEach(res => {
            need[res] = Math.max(0, max[res] - alloc[res]);
        });
        return need;
    }
}

window.bankerRAG = new BankerAndRAG();

class SyncResource {
    constructor(id, name, maxCount = 1) {
        this.id = id;
        this.name = name;
        this.maxCount = maxCount;
        this.availableCount = maxCount;
        this.holderPID = null;
        this.waitingQueue = [];
        this.lockStatus = 'FREE';
    }

    acquire(pid) {
        if (this.holderPID === pid) return true;

        if (this.availableCount > 0 && this.waitingQueue.length === 0) {
            this.availableCount--;
            this.holderPID = pid;
            this.lockStatus = 'LOCKED';
            window.uiController.logSystem(`[MUTEX LOCK] Process ${pid} acquired Mutex Lock for ${this.name}`, 'lock');
            return true;
        } else {
            if (!this.waitingQueue.includes(pid)) {
                this.waitingQueue.push(pid);
                window.uiController.logSystem(`[MUTEX BLOCK] Process ${pid} blocked waiting for ${this.name} Lock`, 'lock');
            }
            return false;
        }
    }

    release(pid) {
        if (this.holderPID === pid) {
            this.holderPID = null;
            this.availableCount++;
            window.uiController.logSystem(`[MUTEX UNLOCK] Process ${pid} released Mutex Lock for ${this.name}`, 'lock');

            if (this.waitingQueue.length > 0) {
                const nextPID = this.waitingQueue.shift();
                this.holderPID = nextPID;
                this.availableCount--;
                this.lockStatus = 'LOCKED';
                window.uiController.logSystem(`[MUTEX HANDOVER] Lock for ${this.name} granted to waiting Process ${nextPID}`, 'lock');
                
                const proc = window.processManager.getProcess(nextPID);
                if (proc && proc.state === 'WAITING') {
                    proc.state = 'READY';
                }
            } else {
                this.lockStatus = 'FREE';
            }
            return true;
        }
        return false;
    }
}

class SynchronizationEngine {
    constructor() {
        this.syncResources = {
            'ICU': new SyncResource('ICU', 'ICU Bed Unit', 1),
            'Ventilator': new SyncResource('Ventilator', 'Ventilator System', 1),
            'MRI': new SyncResource('MRI', 'MRI Scanner Suite', 1),
            'OR': new SyncResource('OR', 'Operating Room Suite', 1)
        };
    }

    requestResource(resourceKey, pid) {
        if (!this.syncResources[resourceKey]) return true;
        return this.syncResources[resourceKey].acquire(pid);
    }

    releaseResourceByPID(pid) {
        Object.values(this.syncResources).forEach(res => {
            if (res.holderPID === pid) {
                res.release(pid);
            }
            const idx = res.waitingQueue.indexOf(pid);
            if (idx !== -1) {
                res.waitingQueue.splice(idx, 1);
            }
        });
    }

    isHolder(resourceKey, pid) {
        const res = this.syncResources[resourceKey];
        return res ? res.holderPID === pid : false;
    }

    reset() {
        this.syncResources = {
            'ICU': new SyncResource('ICU', 'ICU Bed Unit', 1),
            'Ventilator': new SyncResource('Ventilator', 'Ventilator System', 1),
            'MRI': new SyncResource('MRI', 'MRI Scanner Suite', 1),
            'OR': new SyncResource('OR', 'Operating Room Suite', 1)
        };
    }
}

class MemoryManager {
    constructor() {
        this.totalMemory = 1024;
        this.allocationPolicy = 'first_fit';
        this.memoryBlocks = [
            { id: 'block-0', start: 0, size: 1024, isAllocated: false, pid: null, procName: 'Free Space' }
        ];

        this.pageReplacementAlgo = 'lru';
        this.pageFrames = [
            { frameId: 0, pageNum: null, lastAccess: -1, loadTime: -1 },
            { frameId: 1, pageNum: null, lastAccess: -1, loadTime: -1 },
            { frameId: 2, pageNum: null, lastAccess: -1, loadTime: -1 },
            { frameId: 3, pageNum: null, lastAccess: -1, loadTime: -1 }
        ];

        this.pageHits = 0;
        this.pageFaults = 0;
        this.accessCounter = 0;
    }

    allocateMemory(pid, procName, sizeKB) {
        let candidateIdx = -1;
        const freeBlocks = [];
        this.memoryBlocks.forEach((blk, idx) => {
            if (!blk.isAllocated && blk.size >= sizeKB) {
                freeBlocks.push({ idx: idx, block: blk });
            }
        });

        if (freeBlocks.length === 0) {
            window.uiController.logSystem(`[MEMORY ERROR] Out of Memory! Cannot allocate ${sizeKB}KB for ${pid}`, 'memory');
            return false;
        }

        if (this.allocationPolicy === 'first_fit') {
            candidateIdx = freeBlocks[0].idx;
        } else if (this.allocationPolicy === 'best_fit') {
            freeBlocks.sort((a, b) => a.block.size - b.block.size);
            candidateIdx = freeBlocks[0].idx;
        } else if (this.allocationPolicy === 'worst_fit') {
            freeBlocks.sort((a, b) => b.block.size - a.block.size);
            candidateIdx = freeBlocks[0].idx;
        }

        const targetBlock = this.memoryBlocks[candidateIdx];
        const remainingSize = targetBlock.size - sizeKB;

        targetBlock.isAllocated = true;
        targetBlock.pid = pid;
        targetBlock.procName = procName;
        targetBlock.size = sizeKB;

        if (remainingSize > 0) {
            const newFreeBlock = {
                id: `block-${Date.now()}`,
                start: targetBlock.start + sizeKB,
                size: remainingSize,
                isAllocated: false,
                pid: null,
                procName: 'Free Space'
            };
            this.memoryBlocks.splice(candidateIdx + 1, 0, newFreeBlock);
        }

        window.uiController.logSystem(`[MEMORY ALLOCATED] ${sizeKB}KB assigned to ${pid} using ${this.allocationPolicy.toUpperCase()}`, 'memory');
        return true;
    }

    deallocateMemory(pid) {
        let freed = false;
        this.memoryBlocks.forEach(blk => {
            if (blk.pid === pid) {
                blk.isAllocated = false;
                blk.pid = null;
                blk.procName = 'Free Space';
                freed = true;
            }
        });

        if (freed) {
            this.mergeAdjacentFreeBlocks();
            window.uiController.logSystem(`[MEMORY RECLAIMED] Memory freed for discharged process ${pid}`, 'memory');
        }
    }

    mergeAdjacentFreeBlocks() {
        for (let i = 0; i < this.memoryBlocks.length - 1; i++) {
            const curr = this.memoryBlocks[i];
            const next = this.memoryBlocks[i + 1];

            if (!curr.isAllocated && !next.isAllocated) {
                curr.size += next.size;
                this.memoryBlocks.splice(i + 1, 1);
                i--;
            }
        }
    }

    accessPage(pageNum) {
        this.accessCounter++;
        const pNum = parseInt(pageNum);

        const hitFrame = this.pageFrames.find(f => f.pageNum === pNum);

        if (hitFrame) {
            this.pageHits++;
            hitFrame.lastAccess = this.accessCounter;
            return { hit: true, frameId: hitFrame.frameId, message: `PAGE HIT! Page #${pNum} found in Frame ${hitFrame.frameId}` };
        }

        this.pageFaults++;

        const emptyFrame = this.pageFrames.find(f => f.pageNum === null);
        if (emptyFrame) {
            emptyFrame.pageNum = pNum;
            emptyFrame.lastAccess = this.accessCounter;
            emptyFrame.loadTime = this.accessCounter;
            return { hit: false, frameId: emptyFrame.frameId, message: `PAGE FAULT! Loaded Page #${pNum} into empty Frame ${emptyFrame.frameId}` };
        }

        let victimFrame = null;
        if (this.pageReplacementAlgo === 'lru') {
            this.pageFrames.sort((a, b) => a.lastAccess - b.lastAccess);
            victimFrame = this.pageFrames[0];
        } else {
            this.pageFrames.sort((a, b) => a.loadTime - b.loadTime);
            victimFrame = this.pageFrames[0];
        }

        const oldPage = victimFrame.pageNum;
        victimFrame.pageNum = pNum;
        victimFrame.lastAccess = this.accessCounter;
        victimFrame.loadTime = this.accessCounter;

        this.pageFrames.sort((a, b) => a.frameId - b.frameId);

        return { hit: false, frameId: victimFrame.frameId, message: `PAGE FAULT & REPLACEMENT! Evicted Page #${oldPage} from Frame ${victimFrame.frameId} -> Loaded Page #${pNum}` };
    }

    getStats() {
        let used = 0;
        this.memoryBlocks.forEach(b => { if (b.isAllocated) used += b.size; });
        const free = this.totalMemory - used;
        let extFrag = 0;
        this.memoryBlocks.forEach(b => { if (!b.isAllocated && b.size < 64) extFrag += b.size; });

        return { total: this.totalMemory, used: used, free: free, extFrag: extFrag };
    }

    reset() {
        this.memoryBlocks = [
            { id: 'block-0', start: 0, size: 1024, isAllocated: false, pid: null, procName: 'Free Space' }
        ];
        this.pageFrames = [
            { frameId: 0, pageNum: null, lastAccess: -1, loadTime: -1 },
            { frameId: 1, pageNum: null, lastAccess: -1, loadTime: -1 },
            { frameId: 2, pageNum: null, lastAccess: -1, loadTime: -1 },
            { frameId: 3, pageNum: null, lastAccess: -1, loadTime: -1 }
        ];
        this.pageHits = 0;
        this.pageFaults = 0;
        this.accessCounter = 0;
    }
}

class OSEngine {
    constructor() {
        this.isRunning = false;
        this.tickCounter = 0;
        this.tickSpeedMs = 800;
        this.timerInterval = null;
    }

    start() {
        if (this.isRunning) return;
        this.isRunning = true;
        window.uiController.updateStatusUI('RUNNING');

        this.timerInterval = setInterval(() => {
            this.executeClockTick();
        }, this.tickSpeedMs);

        window.uiController.logSystem('[SIMULATION STARTED] Clock ticks active.', 'system');
    }

    pause() {
        if (!this.isRunning) return;
        this.isRunning = false;
        clearInterval(this.timerInterval);
        this.timerInterval = null;
        window.uiController.updateStatusUI('PAUSED');
        window.uiController.logSystem('[SIMULATION PAUSED] Clock ticks suspended.', 'system');
    }

    step() {
        this.pause();
        this.executeClockTick();
        window.uiController.logSystem(`[SIMULATION STEP] Executed Tick ${this.tickCounter}`, 'system');
    }

    setSpeed(speedMs) {
        this.tickSpeedMs = parseInt(speedMs);
        if (this.isRunning) {
            this.pause();
            this.start();
        }
    }

    executeClockTick() {
        this.tickCounter++;
        const allProcs = window.processManager.getAllProcesses();

        const readyQueue = [];
        allProcs.forEach(proc => {
            if (proc.state === 'NEW') {
                const allocSuccess = window.memoryManager.allocateMemory(proc.pid, proc.name, proc.memoryRequired);
                if (allocSuccess) {
                    proc.state = 'READY';
                }
            }

            if (proc.state === 'READY' || proc.state === 'RUNNING') {
                if (proc.resourceClaim && proc.resourceClaim !== 'none') {
                    const lockGranted = window.syncEngine.requestResource(proc.resourceClaim, proc.pid);
                    if (lockGranted) {
                        readyQueue.push(proc);
                    } else {
                        proc.state = 'WAITING';
                    }
                } else {
                    readyQueue.push(proc);
                }
            }
        });

        window.cpuScheduler.scheduleNext(this.tickCounter, readyQueue, window.processManager);
        window.uiController.renderAll();
    }

    loadPreset(presetKey) {
        this.pause();
        window.processManager.resetAll();
        window.cpuScheduler.reset();

            'ICU': { total: 3, available: 3 },

            'OR': { total: 2, available: 2 }
        };
        window.syncEngine.reset();
        window.memoryManager.reset();
        this.tickCounter = 0;

        if (presetKey === 'default') {
            window.processManager.createProcess('Cardiac Emergency', 1, 6, 128, 'ICU');
            window.processManager.createProcess('Appendectomy Surgery', 2, 5, 96, 'OR');
            window.processManager.createProcess('Brain MRI Scan', 3, 4, 64, 'MRI');
            window.processManager.createProcess('Routine Blood Test', 4, 3, 32, 'none');
            window.uiController.logSystem('[PRESET LOADED] Balanced ER Load (P1, P2, P3, P4 cases added)', 'system');

        } else if (presetKey === 'mass_casualty') {
            window.cpuScheduler.setAlgorithm('priority_p');
            document.getElementById('algo-select').value = 'priority_p';
            
            window.processManager.createProcess('Minor Wound Care', 4, 8, 32, 'none');
            window.processManager.createProcess('Trauma - Car Crash 1', 1, 5, 128, 'ICU');
            window.processManager.createProcess('Trauma - Car Crash 2', 1, 4, 128, 'Ventilator');
            window.processManager.createProcess('Severe Burn Case', 1, 6, 96, 'OR');
            window.uiController.logSystem('[PRESET LOADED] Mass Casualty Incident (Preemptive Priority Preemption active)', 'system');

        } else if (presetKey === 'deadlock_risk') {
            window.processManager.createProcess('Organ Transplant (P101)', 1, 8, 128, 'OR');
            window.processManager.createProcess('High Risk ICU (P102)', 2, 7, 96, 'Ventilator');
            window.uiController.logSystem('[PRESET LOADED] Resource Contention (Simulates multi-resource lock deadlock risk)', 'system');
